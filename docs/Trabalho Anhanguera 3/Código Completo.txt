<script>
  const escolhas = ['Pedra', 'Papel', 'Tesoura'];

  document.querySelectorAll('button').forEach(botao => {
    botao.addEventListener('click', function() {
      const jogador = parseInt(this.dataset.jogada);
      const computador = Math.floor(Math.random() * escolhas.length);

      let resultado = "";

      if ((jogador === 0 && computador === 1) ||
              (jogador === 1 && computador === 2) ||
              (jogador === 2 && computador === 0)) {
        resultado = `Você perdeu! Você escolheu ${escolhas[jogador]} e o computador ${escolhas[computador]}.`;
      }
      if ((jogador === 2 && computador === 1) ||
              (jogador === 1 && computador === 0) ||
              (jogador === 0 && computador === 2)) {
        resultado = `Você ganhou! Você escolheu ${escolhas[jogador]} e o computador ${escolhas[computador]}.`;
      }
      if (jogador === computador) {
        resultado = `Parece que empatou! Você escolheu ${escolhas[jogador]} e o computador ${escolhas[computador]}.`;
      }
      document.getElementById('resultado').textContent = resultado;
      console.log(resultado);
    });
  });
</script>
